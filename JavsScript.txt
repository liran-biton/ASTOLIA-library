<script>
  /*******************************************************
   *  ASTOLIA LIBRARY — JavaScript.html
   *  Client-side logic: auth, session, tabs, Digital Creator
   *******************************************************/

  const SESSION_KEY = 'astolia_session';
  const ROLES = { CREATOR: 'Content Creator', EDITOR: 'Video Editor', ADMIN: 'Admin' };

  let currentUser = null;   // { email, displayName, role }
  let currentToken = null;
  let searchDebounceTimer = null;

  /* ---------------- google.script.run as Promise ---------------- */
  function callServer(fn, ...args) {
    return new Promise((resolve, reject) => {
      google.script.run
        .withSuccessHandler(resolve)
        .withFailureHandler(err => reject(err))
        [fn](...args);
    });
  }

  /* ------------------------- INIT -------------------------------- */
  window.addEventListener('DOMContentLoaded', () => {
    init();
    const pwField = document.getElementById('passwordInput');
    if (pwField) pwField.addEventListener('keydown', e => { if (e.key === 'Enter') handleLogin(); });
  });

  function init() {
    const raw = localStorage.getItem(SESSION_KEY);
    if (!raw) return; // show login screen (default state)

    let saved;
    try { saved = JSON.parse(raw); } catch (e) { localStorage.removeItem(SESSION_KEY); return; }

    if (!saved.token || !saved.expires || Date.now() > saved.expires) {
      localStorage.removeItem(SESSION_KEY);
      return;
    }

    currentToken = saved.token;
    callServer('validateSession', currentToken)
      .then(user => {
        if (!user) { localStorage.removeItem(SESSION_KEY); return; }
        currentUser = user;
        enterApp();
      })
      .catch(() => localStorage.removeItem(SESSION_KEY));
  }

  /* ------------------------- LOGIN FLOW --------------------------- */

  function setLoginMsg(text, type) {
    const el = document.getElementById('loginMsg');
    el.textContent = text || '';
    el.className = 'msg' + (type ? ' ' + type : '');
  }

  function handleLogin() {
    const username = document.getElementById('usernameInput').value.trim();
    const password = document.getElementById('passwordInput').value;

    if (!username || !password) {
      setLoginMsg('Please enter username and password.', 'error');
      return;
    }

    const btn = document.getElementById('loginBtn');
    btn.disabled = true;
    btn.innerHTML = '<span class="spinner"></span>Signing in...';
    setLoginMsg('', '');

    callServer('login', username, password)
      .then(res => {
        btn.disabled = false;
        btn.textContent = 'Sign In';
        if (res.success) {
          currentToken = res.token;
          currentUser = res.user;
          localStorage.setItem(SESSION_KEY, JSON.stringify({ token: res.token, expires: res.expires }));
          enterApp();
        } else {
          setLoginMsg(res.message, 'error');
        }
      })
      .catch(() => {
        btn.disabled = false;
        btn.textContent = 'Sign In';
        setLoginMsg('Something went wrong. Please try again.', 'error');
      });
  }

  function handleLogout() {
    callServer('logoutSession', currentToken).finally(() => {
      localStorage.removeItem(SESSION_KEY);
      currentToken = null;
      currentUser = null;
      document.getElementById('appShell').classList.remove('active');
      document.getElementById('loginScreen').style.display = 'flex';
      document.getElementById('usernameInput').value = '';
      document.getElementById('passwordInput').value = '';
      setLoginMsg('', '');
    });
  }

  /* ------------------------- APP SHELL ----------------------------- */

  function enterApp() {
    document.getElementById('loginScreen').style.display = 'none';
    document.getElementById('appShell').classList.add('active');

    document.getElementById('userChipName').textContent = currentUser.displayName || currentUser.email;
    document.getElementById('userChipRole').textContent = currentUser.role;

    applyRoleVisibility();
    switchTab('digitalCreator');
    refreshPendingBadge();
  }

  function applyRoleVisibility() {
    const role = currentUser.role;
    const libraryTab = document.querySelector('.nav-item[data-tab="library"]');
    const partnershipsTab = document.querySelector('.nav-item[data-tab="partnerships"]');

    const privileged = (role === ROLES.EDITOR || role === ROLES.ADMIN);
    libraryTab.style.display = privileged ? 'flex' : 'none';
    partnershipsTab.style.display = privileged ? 'flex' : 'none';

    document.getElementById('newPartnershipBtn').classList.toggle('hidden', !privileged);
  }

  function refreshPendingBadge() {
    if (currentUser.role !== ROLES.CREATOR) return;
    callServer('getPendingApprovalCount', currentToken).then(count => {
      const badge = document.getElementById('expBadge');
      if (count > 0) {
        badge.textContent = count;
        badge.classList.remove('hidden');
      } else {
        badge.classList.add('hidden');
      }
    }).catch(() => {});
  }

  function switchTab(tab) {
    document.querySelectorAll('.nav-item').forEach(el => el.classList.remove('active'));
    const navEl = document.querySelector('.nav-item[data-tab="' + tab + '"]');
    if (navEl) navEl.classList.add('active');

    document.querySelectorAll('.tab-page').forEach(el => el.style.display = 'none');
    const pageEl = document.getElementById('page-' + tab);
    if (pageEl) pageEl.style.display = 'block';

    if (tab === 'digitalCreator') {
      loadCities();
      loadCreators();
    }
    if (tab === 'partnerships') {
      switchPartnershipSubTab('list');
    }
    if (tab === 'experiences') {
      loadExperiences();
    }
    if (tab === 'library') {
      loadLibraryFilterOptions();
      loadLibrary();
    }
  }

  /* ------------------------- DIGITAL CREATOR ------------------------ */

  function loadCities() {
    callServer('getCitiesList', currentToken).then(cities => {
      const sel = document.getElementById('cityFilter');
      const current = sel.value;
      sel.innerHTML = '<option value="">All cities</option>' +
        cities.map(c => '<option value="' + escapeHtml(c) + '">' + escapeHtml(c) + '</option>').join('');
      sel.value = current;
    }).catch(() => {});
  }

  function debouncedLoadCreators() {
    clearTimeout(searchDebounceTimer);
    searchDebounceTimer = setTimeout(loadCreators, 300);
  }

  function loadCreators() {
    const grid = document.getElementById('creatorsGrid');
    grid.innerHTML = '<div class="empty-state">Loading profiles...</div>';

    const filters = {
      city: document.getElementById('cityFilter').value,
      search: document.getElementById('searchFilter').value
    };

    callServer('getDigitalCreators', currentToken, filters)
      .then(res => {
        document.getElementById('createProfileBtn').classList.toggle('hidden', !res.canCreate);
        renderCreators(res.items);
      })
      .catch(err => {
        grid.innerHTML = '<div class="empty-state">' + escapeHtml((err && err.message) ? err.message : 'Could not load profiles. Please refresh.') + '</div>';
      });
  }

  function renderCreators(items) {
    const grid = document.getElementById('creatorsGrid');
    items = Array.isArray(items) ? items : [];
    if (!items.length) {
      grid.innerHTML = '<div class="empty-state">No profiles found yet.</div>';
      return;
    }

    grid.innerHTML = items.map(it => {
      const initials = (it.name || '?').trim().split(/\s+/).map(w => w[0]).slice(0, 2).join('').toUpperCase();
      const socials = [];
      if (it.instagram) socials.push('<a class="social-link" href="' + escapeAttr(it.instagram) + '" target="_blank">Instagram</a>');
      if (it.facebook)  socials.push('<a class="social-link" href="' + escapeAttr(it.facebook)  + '" target="_blank">Facebook</a>');
      if (it.youtube)   socials.push('<a class="social-link" href="' + escapeAttr(it.youtube)   + '" target="_blank">YouTube</a>');

      return '' +
        '<div class="creator-card">' +
          '<div class="avatar-ring">' + escapeHtml(initials) + '</div>' +
          '<div class="creator-name">' + escapeHtml(it.name) + '</div>' +
          (it.city ? '<div class="creator-city">' + escapeHtml(it.city) + '</div>' : '') +
          (it.about ? '<div class="creator-about">' + escapeHtml(it.about) + '</div>' : '') +
          '<div class="creator-contact">' + escapeHtml(it.email) + (it.phone ? ' · ' + escapeHtml(it.phone) : '') + '</div>' +
          (socials.length ? '<div class="social-row">' + socials.join('') + '</div>' : '') +
          '<div class="card-footer">' +
            (it.canEdit ? '<button class="edit-btn" onclick=\'openEditModal(' + JSON.stringify(it).replace(/'/g, "&#39;") + ')\'>Edit profile</button>' : '') +
          '</div>' +
        '</div>';
    }).join('');
  }

  /* ------------------------- PROFILE MODAL --------------------------- */

  function openCreateModal() {
    document.getElementById('modalTitle').textContent = 'Create Profile';
    document.getElementById('profileId').value = '';
    ['fName','fPhone','fCity','fAbout','fInstagram','fFacebook','fYoutube'].forEach(id => document.getElementById(id).value = '');
    document.getElementById('modalMsg').textContent = '';

    const emailWrap = document.getElementById('emailFieldWrap');
    const emailInput = document.getElementById('fEmail');
    if (currentUser.role === ROLES.CREATOR) {
      emailInput.value = currentUser.email;
      emailWrap.style.opacity = '0.6';
      emailInput.disabled = true;
    } else {
      emailInput.value = '';
      emailWrap.style.opacity = '1';
      emailInput.disabled = false;
    }

    document.getElementById('profileModal').classList.add('active');
  }

  function openEditModal(item) {
    document.getElementById('modalTitle').textContent = 'Edit Profile';
    document.getElementById('profileId').value = item.id;
    document.getElementById('fName').value = item.name || '';
    document.getElementById('fPhone').value = item.phone || '';
    document.getElementById('fCity').value = item.city || '';
    document.getElementById('fAbout').value = item.about || '';
    document.getElementById('fInstagram').value = item.instagram || '';
    document.getElementById('fFacebook').value = item.facebook || '';
    document.getElementById('fYoutube').value = item.youtube || '';
    document.getElementById('modalMsg').textContent = '';

    const emailWrap = document.getElementById('emailFieldWrap');
    const emailInput = document.getElementById('fEmail');
    emailInput.value = item.email || '';
    if (currentUser.role === ROLES.CREATOR) {
      emailWrap.style.opacity = '0.6';
      emailInput.disabled = true;
    } else {
      emailWrap.style.opacity = '1';
      emailInput.disabled = false;
    }

    document.getElementById('profileModal').classList.add('active');
  }

  function closeModal() {
    document.getElementById('profileModal').classList.remove('active');
  }

  function handleSaveProfile() {
    const id = document.getElementById('profileId').value;
    const data = {
      name: document.getElementById('fName').value.trim(),
      email: document.getElementById('fEmail').value.trim(),
      phone: document.getElementById('fPhone').value.trim(),
      city: document.getElementById('fCity').value.trim(),
      about: document.getElementById('fAbout').value.trim(),
      instagram: document.getElementById('fInstagram').value.trim(),
      facebook: document.getElementById('fFacebook').value.trim(),
      youtube: document.getElementById('fYoutube').value.trim()
    };

    if (!data.name) {
      document.getElementById('modalMsg').textContent = 'Name is required.';
      return;
    }

    const btn = document.getElementById('saveProfileBtn');
    btn.disabled = true;
    btn.innerHTML = '<span class="spinner"></span>Saving...';

    const action = id
      ? callServer('updateDigitalCreatorProfile', currentToken, id, data)
      : callServer('createDigitalCreatorProfile', currentToken, data);

    action
      .then(() => {
        btn.disabled = false;
        btn.textContent = 'Save Profile';
        closeModal();
        loadCities();
        loadCreators();
      })
      .catch(err => {
        btn.disabled = false;
        btn.textContent = 'Save Profile';
        document.getElementById('modalMsg').textContent = (err && err.message) ? err.message : 'Could not save profile.';
      });
  }

  /* ------------------------- CREATOR PARTNERSHIPS -------------------- */

  function switchPartnershipSubTab(sub) {
    document.querySelectorAll('.sub-tab').forEach(el => el.classList.remove('active'));
    const tabEl = document.querySelector('.sub-tab[data-subtab="' + sub + '"]');
    if (tabEl) tabEl.classList.add('active');

    document.querySelectorAll('.subtab-page').forEach(el => el.classList.add('hidden'));
    document.getElementById('subtab-' + sub).classList.remove('hidden');

    if (sub === 'list') loadPartnerships();
    if (sub === 'branches') loadBranches();
    if (sub === 'benefits') loadBenefitsAdmin();
  }

  function clearPartnershipDate() {
    document.getElementById('pDateFilter').value = '';
    loadPartnerships();
  }

  function loadPartnerships() {
    const grid = document.getElementById('partnershipsGrid');
    grid.innerHTML = '<div class="empty-state">Loading partnerships...</div>';

    const filters = {
      date: document.getElementById('pDateFilter').value,
      statusFilter: document.getElementById('pStatusFilter').value
    };

    callServer('getPartnerships', currentToken, filters)
      .then(items => {
        try {
          renderPartnerships(items, grid, false);
        } catch (renderErr) {
          console.error('renderPartnerships failed', renderErr, items);
          grid.innerHTML = '<div class="empty-state">Display error: ' + escapeHtml(renderErr.message) + '</div>';
        }
      })
      .catch(err => { grid.innerHTML = '<div class="empty-state">' + escapeHtml((err && err.message) ? err.message : 'Could not load partnerships.') + '</div>'; });
  }

  function renderPartnerships(items, grid, isExperiencesView) {
    items = Array.isArray(items) ? items : [];
    if (!items.length) {
      grid.innerHTML = '<div class="empty-state">No partnerships found.</div>';
      return;
    }

    grid.innerHTML = items.map(it => {
      const statusClass = it.status === 'Approved' ? 'status-approved'
        : it.status === 'Cancelled' ? 'status-cancelled' : 'status-pending';

      const contentTags = [];
      if (it.posts > 0) contentTags.push(it.posts + ' Post' + (it.posts > 1 ? 's' : ''));
      if (it.reels > 0) contentTags.push(it.reels + ' Reels');
      if (it.stories > 0) contentTags.push(it.stories + ' Story');

      const benefitTags = (it.benefits || '').split(',').map(b => b.trim()).filter(Boolean);

      let actions = '';
      if (it.canApprove) {
        actions += '<button class="edit-btn" onclick=\'openApproveModal(' + JSON.stringify(it).replace(/'/g, "&#39;") + ')\'>Review &amp; Approve</button>';
      }
      if (it.canUploadContent) {
        actions += '<button class="edit-btn" onclick=\'openUploadModal(' + JSON.stringify(it).replace(/'/g, "&#39;") + ')\'>Upload Content</button>';
      }
      if (it.canCancel) {
        actions += '<button class="edit-btn" style="color:var(--danger);margin-left:12px;" onclick="handleCancelPartnership(\'' + it.id + '\')">Cancel</button>';
      }

      const showContentProgress = it.status === 'Approved';

      return '' +
        '<div class="creator-card">' +
          '<div style="display:flex;justify-content:space-between;align-items:flex-start;">' +
            '<div class="creator-name">' + escapeHtml(it.creatorName) + '</div>' +
            '<span class="status-pill ' + statusClass + '">' + escapeHtml(it.status) + '</span>' +
          '</div>' +
          '<div class="creator-city">' + escapeHtml(it.branchName) + '</div>' +
          '<div class="info-row"><span>Date</span><span>' + formatDateUS(it.visitDate) + (it.visitTime ? ' · ' + escapeHtml(it.visitTime) : '') + '</span></div>' +
          '<div class="info-row"><span>People</span><span>' + escapeHtml(String(it.peopleCount)) + '</span></div>' +
          '<div class="info-row"><span>Upload deadline</span><span>' + escapeHtml(it.uploadDeadline) + '</span></div>' +
          (benefitTags.length ? '<div class="tag-row">' + benefitTags.map(b => '<span class="tag">' + escapeHtml(b) + '</span>').join('') + '</div>' : '') +
          (contentTags.length ? '<div class="tag-row">' + contentTags.map(c => '<span class="tag">' + escapeHtml(c) + '</span>').join('') + '</div>' : '') +
          (it.status === 'Cancelled' && it.cancelledBy ? '<div class="info-row"><span>Cancelled by</span><span>' + escapeHtml(it.cancelledBy) + '</span></div>' : '') +
          (it.status === 'Approved' && it.pregnancyDeclaration ? '<div class="info-row"><span>Pregnancy declaration</span><span>' + escapeHtml(it.pregnancyDeclaration) + '</span></div>' : '') +
          (showContentProgress ? '<div class="content-progress">Content uploaded: <b>' + it.contentImages + '/' + it.maxImages + '</b> photos, <b>' + it.contentVideos + '/' + it.maxVideos + '</b> videos</div>' : '') +
          (actions ? '<div class="card-footer">' + actions + '</div>' : '') +
        '</div>';
    }).join('');
  }

  function openCreatePartnershipModal() {
    document.getElementById('npMsg').textContent = '';
    document.getElementById('npDate').value = '';
    document.getElementById('npTime').value = '';
    document.getElementById('npPeople').value = '1';
    document.getElementById('npDeadline').value = '';
    ['npPostChk','npReelsChk','npStoryChk'].forEach(id => document.getElementById(id).checked = false);
    ['npPostQty','npReelsQty','npStoryQty'].forEach(id => { document.getElementById(id).value = ''; document.getElementById(id).disabled = true; });

    Promise.all([
      callServer('getContentCreatorsList', currentToken),
      callServer('getActiveSpaBranchesList', currentToken),
      callServer('getActiveBenefitsList', currentToken)
    ]).then(([creators, branches, benefits]) => {
      creators = Array.isArray(creators) ? creators : [];
      branches = Array.isArray(branches) ? branches : [];
      benefits = Array.isArray(benefits) ? benefits : [];

      document.getElementById('npCreator').innerHTML = '<option value="" disabled selected>Select a content creator...</option>' +
        (creators.length
          ? creators.map(c => '<option value="' + escapeHtml(c.email) + '">' + escapeHtml(c.name) + '</option>').join('')
          : '');

      document.getElementById('npBranch').innerHTML = '<option value="" disabled selected>Select a spa branch...</option>' +
        (branches.length
          ? branches.map(b => '<option value="' + escapeHtml(b.id) + '">' + escapeHtml(b.name) + '</option>').join('')
          : '');

      const list = document.getElementById('npBenefitsList');
      list.innerHTML = benefits.length
        ? benefits.map(b => '<label><input type="checkbox" class="npBenefitChk" value="' + escapeHtml(b.treatment + ' (' + b.duration + ')') + '"> ' + escapeHtml(b.treatment) + ' — ' + escapeHtml(b.duration) + '</label>').join('')
        : '<div class="empty-state" style="padding:20px;">No active benefits defined yet.</div>';

      document.getElementById('partnershipModal').classList.add('active');
    }).catch(err => {
      document.getElementById('npMsg').textContent = (err && err.message) ? err.message : 'Could not load form data. Please try again.';
    });
  }

  function toggleQtyField(chkId, qtyId) {
    const chk = document.getElementById(chkId);
    const qty = document.getElementById(qtyId);
    qty.disabled = !chk.checked;
    if (!chk.checked) qty.value = '';
    else if (!qty.value) qty.value = '1';
  }

  function closePartnershipModal() {
    document.getElementById('partnershipModal').classList.remove('active');
  }

  function handleCreatePartnership() {
    const benefits = Array.from(document.querySelectorAll('.npBenefitChk:checked')).map(el => el.value);

    const data = {
      creatorEmail: document.getElementById('npCreator').value,
      branchId: document.getElementById('npBranch').value,
      peopleCount: document.getElementById('npPeople').value,
      visitDate: document.getElementById('npDate').value,
      visitTime: document.getElementById('npTime').value,
      benefits: benefits,
      posts: document.getElementById('npPostChk').checked ? (document.getElementById('npPostQty').value || 0) : 0,
      reels: document.getElementById('npReelsChk').checked ? (document.getElementById('npReelsQty').value || 0) : 0,
      stories: document.getElementById('npStoryChk').checked ? (document.getElementById('npStoryQty').value || 0) : 0,
      uploadDeadline: document.getElementById('npDeadline').value
    };

    if (!data.creatorEmail) { document.getElementById('npMsg').textContent = 'Please select a content creator.'; return; }
    if (!data.branchId) { document.getElementById('npMsg').textContent = 'Please select a spa branch.'; return; }
    if (!data.visitDate) { document.getElementById('npMsg').textContent = 'Please select a visit date.'; return; }
    if (!data.uploadDeadline) { document.getElementById('npMsg').textContent = 'Please select an upload deadline.'; return; }

    const btn = document.getElementById('npSaveBtn');
    btn.disabled = true;
    btn.innerHTML = '<span class="spinner"></span>Sending...';

    callServer('createPartnership', currentToken, data)
      .then(() => {
        btn.disabled = false;
        btn.textContent = 'Send to Creator';
        closePartnershipModal();
        loadPartnerships();
      })
      .catch(err => {
        btn.disabled = false;
        btn.textContent = 'Send to Creator';
        document.getElementById('npMsg').textContent = (err && err.message) ? err.message : 'Could not create partnership.';
      });
  }

  function handleCancelPartnership(id) {
    if (!confirm('Cancel this partnership? The content creator will be notified.')) return;
    callServer('cancelPartnership', currentToken, id)
      .then(() => {
        loadPartnerships();
        loadExperiences();
      })
      .catch(err => alert((err && err.message) ? err.message : 'Could not cancel partnership.'));
  }

  /* ---------------------- SPA BRANCHES (admin/editor) ----------------- */

  function loadBranches() {
    const grid = document.getElementById('branchesGrid');
    grid.innerHTML = '<div class="empty-state">Loading branches...</div>';

    callServer('getSpaBranches', currentToken).then(items => {
      items = Array.isArray(items) ? items : [];
      if (!items.length) { grid.innerHTML = '<div class="empty-state">No spa branches added yet.</div>'; return; }

      grid.innerHTML = items.map(b => {
        const activities = [];
        if (b.swedishMassage) activities.push('Swedish Massage');
        if (b.reflexology) activities.push('Reflexology');
        if (b.poolArea) activities.push('Pool Area');
        if (b.jacuzzi) activities.push('Jacuzzi');
        if (b.sauna) activities.push('Sauna');

        return '' +
          '<div class="creator-card">' +
            '<div style="display:flex;justify-content:space-between;align-items:flex-start;">' +
              '<div class="creator-name">' + escapeHtml(b.name) + '</div>' +
              '<span class="status-pill ' + (b.status === 'Active' ? 'status-active' : 'status-inactive') + '">' + escapeHtml(b.status) + '</span>' +
            '</div>' +
            (b.address ? '<div class="creator-about">' + escapeHtml(b.address) + '</div>' : '') +
            (b.hours ? '<div class="info-row"><span>Hours</span><span>' + escapeHtml(b.hours) + '</span></div>' : '') +
            (b.contact ? '<div class="info-row"><span>Contact</span><span>' + escapeHtml(b.contact) + '</span></div>' : '') +
            (activities.length ? '<div class="tag-row">' + activities.map(a => '<span class="tag">' + escapeHtml(a) + '</span>').join('') + '</div>' : '') +
            '<div class="card-footer"><button class="edit-btn" onclick=\'openBranchModal(' + JSON.stringify(b).replace(/'/g, "&#39;") + ')\'>Edit</button></div>' +
          '</div>';
      }).join('');
    }).catch(err => { grid.innerHTML = '<div class="empty-state">' + escapeHtml((err && err.message) ? err.message : 'Could not load branches.') + '</div>'; });
  }

  function openBranchModal(branch) {
    document.getElementById('branchMsg').textContent = '';
    document.getElementById('branchModalTitle').textContent = branch ? 'Edit Spa Branch' : 'Add Spa Branch';
    document.getElementById('branchId').value = branch ? branch.id : '';
    document.getElementById('brName').value = branch ? branch.name : '';
    document.getElementById('brAddress').value = branch ? branch.address : '';
    document.getElementById('brHours').value = branch ? branch.hours : '';
    document.getElementById('brContact').value = branch ? branch.contact : '';
    document.getElementById('brSwedish').checked = branch ? !!branch.swedishMassage : false;
    document.getElementById('brReflex').checked = branch ? !!branch.reflexology : false;
    document.getElementById('brPool').checked = branch ? !!branch.poolArea : false;
    document.getElementById('brJacuzzi').checked = branch ? !!branch.jacuzzi : false;
    document.getElementById('brSauna').checked = branch ? !!branch.sauna : false;
    document.getElementById('brStatus').value = branch ? branch.status : 'Active';
    document.getElementById('branchModal').classList.add('active');
  }

  function closeBranchModal() {
    document.getElementById('branchModal').classList.remove('active');
  }

  function handleSaveBranch() {
    const id = document.getElementById('branchId').value;
    const data = {
      name: document.getElementById('brName').value.trim(),
      address: document.getElementById('brAddress').value.trim(),
      hours: document.getElementById('brHours').value.trim(),
      contact: document.getElementById('brContact').value.trim(),
      swedishMassage: document.getElementById('brSwedish').checked,
      reflexology: document.getElementById('brReflex').checked,
      poolArea: document.getElementById('brPool').checked,
      jacuzzi: document.getElementById('brJacuzzi').checked,
      sauna: document.getElementById('brSauna').checked,
      status: document.getElementById('brStatus').value
    };
    if (!data.name) { document.getElementById('branchMsg').textContent = 'Branch name is required.'; return; }

    const btn = document.getElementById('branchSaveBtn');
    btn.disabled = true;
    btn.innerHTML = '<span class="spinner"></span>Saving...';

    const action = id
      ? callServer('updateSpaBranch', currentToken, id, data)
      : callServer('createSpaBranch', currentToken, data);

    action.then(() => {
      btn.disabled = false;
      btn.textContent = 'Save Branch';
      closeBranchModal();
      loadBranches();

  function renderStars(score) {
    score = Math.max(0, Math.min(5, Number(score) || 0));
    let html = '<span class="ai-stars">';
    for (let i = 1; i <= 5; i++) {

    html += '</span>';

  }

  function fileToBase64(file) {
    return new Promise((resolve, reject) => {
      const reader = new FileReader();
      reader.onload = () => resolve(String(reader.result).split(',')[1] || '');
      reader.onerror = () => reject(new Error('Could not read file: ' + file.name));
      reader.readAsDataURL(file);
    });
  }

  // Captures 3 frames (25%/50%/75% through the clip) from a video File,
  // entirely in the browser, so the AI can analyze a video's content
  // without Apps Script ever needing to decode video itself.
  function extractVideoFrames(file) {
    return new Promise((resolve, reject) => {
      const video = document.createElement('video');
      video.preload = 'auto';
      video.muted = true;
      video.playsInline = true;
      video.style.position = 'fixed';
      video.style.left = '-9999px';
      document.body.appendChild(video);

      const objectUrl = URL.createObjectURL(file);
      video.src = objectUrl;

      const fractions = [0.25, 0.5, 0.75];
      const frames = [];
      let idx = 0;

      function cleanup() {
        URL.revokeObjectURL(objectUrl);
        video.remove();
      }

      function captureAt(fraction) {
        video.currentTime = Math.max(0, Math.min(video.duration * fraction, video.duration - 0.05));
      }

      video.addEventListener('loadedmetadata', () => {
        if (!isFinite(video.duration) || video.duration <= 0) {
          reject(new Error('Could not read video duration for ' + file.name));
          cleanup();
          return;
        }
        captureAt(fractions[0]);
      });

      video.addEventListener('seeked', () => {
        const canvas = document.createElement('canvas');
        canvas.width = video.videoWidth || 640;
        canvas.height = video.videoHeight || 360;
        const ctx = canvas.getContext('2d');
        ctx.drawImage(video, 0, 0, canvas.width, canvas.height);
        frames.push(canvas.toDataURL('image/jpeg', 0.8).split(',')[1] || '');

        idx++;
        if (idx < fractions.length) {
          captureAt(fractions[idx]);
        } else {
          cleanup();
          resolve(frames);
        }
      });

      video.addEventListener('error', () => {
        cleanup();
        reject(new Error('Could not read video: ' + file.name));
      });
    });
  }

  function handleUploadContent() {
    const imageFiles = Array.from(document.getElementById('uploadImagesInput').files || []);
    const videoFiles = Array.from(document.getElementById('uploadVideosInput').files || []);
    const msgEl = document.getElementById('uploadMsg');
    msgEl.textContent = '';

    if (!imageFiles.length && !videoFiles.length) {
      msgEl.textContent = 'Please choose at least one photo or video.';
      return;
    }
    if (imageFiles.length > uploadRemainingImages) {
      msgEl.textContent = 'You can only add ' + uploadRemainingImages + ' more photo(s) for this partnership.';
      return;
    }
    if (videoFiles.length > uploadRemainingVideos) {
      msgEl.textContent = 'You can only add ' + uploadRemainingVideos + ' more video(s) for this partnership.';
      return;
    }

    const MAX_FILE_MB = 25;
    const tooBig = imageFiles.concat(videoFiles).find(f => f.size > MAX_FILE_MB * 1024 * 1024);
    if (tooBig) {
      msgEl.textContent = tooBig.name + ' is larger than ' + MAX_FILE_MB + 'MB — please use a smaller file.';
      return;
    }

    const btn = document.getElementById('uploadSaveBtn');
    btn.disabled = true;
    btn.innerHTML = '<span class="spinner"></span>Preparing files...';
    document.getElementById('uploadProgressList').innerHTML = '<div>Reading files…</div>';

    const allFiles = imageFiles.map(f => ({ file: f, contentType: 'Image' }))
      .concat(videoFiles.map(f => ({ file: f, contentType: 'Video' })));

    Promise.all(allFiles.map(entry => {
      const base64Promise = fileToBase64(entry.file);
      if (entry.contentType === 'Video') {
        return Promise.all([base64Promise, extractVideoFrames(entry.file)]).then(([base64, frames]) => ({
          name: entry.file.name,
          mimeType: entry.file.type || 'application/octet-stream',
          contentType: entry.contentType,
          base64: base64,
          frames: frames
        }));
      }
      return base64Promise.then(base64 => ({
        name: entry.file.name,
        mimeType: entry.file.type || 'application/octet-stream',
        contentType: entry.contentType,
        base64: base64
      }));
    })).then(payload => {
      btn.innerHTML = '<span class="spinner"></span>Uploading & analyzing...';
      document.getElementById('uploadProgressList').innerHTML = '<div>Uploading ' + payload.length + ' file(s) and running AI analysis — this can take a little while…</div>';
      return callServer('uploadPartnershipContent', currentToken, uploadTargetPartnershipId, payload);
    }).then(() => {
      btn.disabled = false;
      btn.textContent = 'Upload';
      closeUploadModal();
      loadExperiences();
    }).catch(err => {
      btn.disabled = false;
      btn.textContent = 'Upload';
      document.getElementById('uploadProgressList').innerHTML = '';
      msgEl.textContent = (err && err.message) ? err.message : 'Could not upload content.';
    });
  }

  function formatDateUS(isoDate) {
    if (!isoDate) return '—';
    const parts = String(isoDate).split('-');
    if (parts.length !== 3) return isoDate;
    return parts[1] + '/' + parts[2] + '/' + parts[0]; // MM/DD/YYYY
  }

  function escapeHtml(str) {
    return String(str == null ? '' : str)
      .replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;')
      .replace(/"/g, '&quot;');
  }
  function escapeAttr(str) {
    let s = String(str == null ? '' : str).trim();
    if (s && !/^https?:\/\//i.test(s)) s = 'https://' + s;
    return escapeHtml(s);
  }
